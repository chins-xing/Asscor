func (r *RateLimiter) Stop() {
    r.mu.Lock()
    defer r.mu.Unlock()
    if r.cleanupTick != nil {
        r.cleanupTick.Stop()
    }
    if !r.stopped {
        r.stopped = true
        close(r.stopCleanup)
    }
}
