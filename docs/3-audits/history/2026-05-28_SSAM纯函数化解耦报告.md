### 纯函数化接入示例

```go
package main

import (
    "fmt"
    ssam "github.com/asscor/ssam"
)

func main() {
    config := ssam.DefaultScoringConfig

    input := ssam.AssessmentInput{
        HostID:      "server-01",
        Threshold:   80,
        ThreatCoeff: 1.0,
        SPCScore:    1.0,
        Checks: []ssam.CheckInput{
            {CheckID: "AS-001", Domain: "attack_surface", Passed: false, Delta: -15},
            {CheckID: "BC-001", Domain: "business_continuity", Passed: false, Delta: -10},
        },
    }

    output, err := ssam.ComputeScore(config, input)
    if err != nil {
        panic(err)
    }
    fmt.Printf("Score: %.2f, Acceptable: %v\n", output.FinalScore, output.Acceptable)
}
```

**无任何依赖。仅需 `go get github.com/asscor/ssam`（或本地 replace）。**
